! R-HQ - Parsa Logistics (Router 2911)
hostname R-HQ
no ip domain-lookup
ip domain-name parsa-log.local
!
enable secret Ena@2026
username netadmin privilege 15 secret Parsa@2026
!
interface GigabitEthernet0/0
 description LAN-HQ
 ip address 10.10.1.1 255.255.255.0
 no shutdown
interface GigabitEthernet0/1
 description Link-to-R-BR
 ip address 172.16.0.1 255.255.255.252
 no shutdown
interface GigabitEthernet0/2
 description Link-to-R-ISP
 ip address 203.0.113.2 255.255.255.252
 no shutdown
!
! Static routes
ip route 10.10.2.0 255.255.255.0 172.16.0.2
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! SSH (run after hostname + domain-name)
crypto key generate rsa modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 2
!
line vty 0 4
 login local
 transport input ssh
 exec-timeout 5 0
end
